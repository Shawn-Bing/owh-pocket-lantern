cmake_minimum_required(VERSION 3.20)
project(OverwatchHelper VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_executable(core-tests tests/core_tests.cpp)
target_include_directories(core-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME core COMMAND core-tests)
if(WIN32)
  add_executable(overwatch-helper WIN32 agent/main.cpp)
  add_executable(owh-demo WIN32 agent/demo.cpp)
  add_executable(platform-tests tests/platform_tests.cpp)
  add_executable(owh-recorder WIN32 recorder/main.cpp)
  add_executable(recorder-tests tests/recorder_tests.cpp)
  add_executable(cn-replay tests/cn_replay.cpp)
  # Build tool, not a test: classifies recorded frames against the profile so a
  # missing step can be told apart from a threshold that is merely too strict.
  add_executable(owh-profile tools/profile_tool.cpp)
  add_test(NAME platform COMMAND platform-tests)
  add_test(NAME recorder COMMAND recorder-tests)
  add_test(NAME cn-recovery COMMAND cn-replay)
  foreach(target overwatch-helper owh-demo owh-recorder platform-tests recorder-tests core-tests cn-replay owh-profile)
    target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
    if(MSVC)
      target_compile_options(${target} PRIVATE /utf-8 /W4 /EHsc)
    else()
      target_link_options(${target} PRIVATE -static)
    endif()
  endforeach()
  if(NOT MSVC)
    target_link_options(overwatch-helper PRIVATE -municode)
    target_link_options(owh-demo PRIVATE -municode)
    target_link_options(owh-recorder PRIVATE -municode)
  endif()
  target_link_libraries(overwatch-helper PRIVATE user32 gdi32 shell32 advapi32 comdlg32 winhttp ole32 dwmapi)
  target_link_libraries(owh-demo PRIVATE user32 gdi32)
  target_link_libraries(platform-tests PRIVATE user32 gdi32 shell32 advapi32 comdlg32 winhttp ole32 dwmapi)
  foreach(target owh-recorder recorder-tests cn-replay owh-profile)
    target_link_libraries(${target} PRIVATE user32 gdi32 shell32 advapi32 comdlg32 winhttp ole32
                          iphlpapi ws2_32 psapi wevtapi gdiplus uuid dwmapi)
  endforeach()
endif()
