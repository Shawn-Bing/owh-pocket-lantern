cmake_minimum_required(VERSION 3.20)
project(OverwatchHelper VERSION 0.5.13 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_executable(automation-protocol-tests tests/automation_protocol_tests.cpp)
target_include_directories(automation-protocol-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME automation-protocol COMMAND automation-protocol-tests)
find_program(OWH_GO_EXECUTABLE NAMES go HINTS "${CMAKE_CURRENT_SOURCE_DIR}/build/tools/go/bin")
if(OWH_GO_EXECUTABLE)
  add_test(NAME automation-protocol-server COMMAND ${CMAKE_COMMAND} -E env
    "OWH_PROTOCOL_FIXTURE=$<TARGET_FILE:automation-protocol-tests>"
    "GOCACHE=${CMAKE_CURRENT_BINARY_DIR}/go-cache"
    ${OWH_GO_EXECUTABLE} test -count=1 -run TestAutomationStateWireContract ./...)
  set_tests_properties(automation-protocol-server PROPERTIES
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/server" TIMEOUT 120)
endif()
add_executable(core-tests tests/core_tests.cpp)
target_include_directories(core-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME core COMMAND core-tests)
add_executable(vision-tests tests/vision_tests.cpp)
target_include_directories(vision-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME vision COMMAND vision-tests)
add_executable(deathmatch-tests tests/deathmatch_tests.cpp)
target_include_directories(deathmatch-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME deathmatch COMMAND deathmatch-tests)
add_executable(visual-calibration-tests tests/visual_calibration_tests.cpp)
target_include_directories(visual-calibration-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME visual-calibration COMMAND visual-calibration-tests)
if(WIN32)
  enable_language(RC)
  configure_file(agent/embedded.rc.in embedded.rc @ONLY)
  set_property(SOURCE ${CMAKE_CURRENT_BINARY_DIR}/embedded.rc APPEND PROPERTY OBJECT_DEPENDS
    ${CMAKE_CURRENT_SOURCE_DIR}/scripts/ocr-worker.ps1 ${CMAKE_CURRENT_SOURCE_DIR}/profiles/cn-16x9.json)
  add_executable(overwatch-helper WIN32 agent/main.cpp ${CMAKE_CURRENT_BINARY_DIR}/embedded.rc)
  add_executable(overwatch-helper-simple WIN32 agent/main.cpp ${CMAKE_CURRENT_BINARY_DIR}/embedded.rc)
  target_compile_definitions(overwatch-helper-simple PRIVATE OWH_SIMPLE_EDITION=1)
  # The packaged main application always requests Windows administrator consent.
  if(MSVC)
    target_link_options(overwatch-helper PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
    target_link_options(overwatch-helper-simple PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
  else()
    enable_language(RC)
    target_sources(overwatch-helper PRIVATE agent/overwatch-helper.rc)
    target_sources(overwatch-helper-simple PRIVATE agent/overwatch-helper.rc)
    target_include_directories(overwatch-helper-simple PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/agent)
    target_include_directories(overwatch-helper PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/agent)
    set_property(SOURCE agent/overwatch-helper.rc APPEND PROPERTY OBJECT_DEPENDS
      ${CMAKE_CURRENT_SOURCE_DIR}/agent/overwatch-helper.manifest)
  endif()
  add_executable(owh-demo WIN32 agent/demo.cpp)
  add_executable(embedded-ocr-tests tools/vision_replay.cpp ${CMAKE_CURRENT_BINARY_DIR}/embedded.rc)
  add_test(NAME embedded-ocr COMMAND embedded-ocr-tests --self-test)
  set_tests_properties(embedded-ocr PROPERTIES TIMEOUT 30)
  add_executable(platform-tests tests/platform_tests.cpp)
  add_executable(background-input-tests tests/background_input_tests.cpp)
  add_test(NAME background-input COMMAND background-input-tests)
  set_tests_properties(background-input PROPERTIES TIMEOUT 15)
  add_executable(recovery-lifecycle-tests tests/recovery_lifecycle_tests.cpp)
  add_test(NAME recovery-lifecycle COMMAND recovery-lifecycle-tests)
  add_executable(neac-cleanup-tests tests/neac_cleanup_tests.cpp)
  add_test(NAME neac-cleanup COMMAND neac-cleanup-tests)
  set_tests_properties(neac-cleanup PROPERTIES TIMEOUT 30)
  add_executable(ui-layout-tests tests/ui_layout_tests.cpp)
  add_executable(simple-ui-tests tests/simple_ui_tests.cpp ${CMAKE_CURRENT_BINARY_DIR}/embedded.rc)
  add_test(NAME simple-ui COMMAND simple-ui-tests)
  set_tests_properties(simple-ui PROPERTIES TIMEOUT 30)
  add_test(NAME ui-layout COMMAND ui-layout-tests)
  set_tests_properties(ui-layout PROPERTIES TIMEOUT 30)
  add_executable(owh-recorder WIN32 recorder/main.cpp)
  add_executable(recorder-tests tests/recorder_tests.cpp)
  add_executable(cn-replay tests/cn_replay.cpp)
  # Build tool, not a test: classifies recorded frames against the profile so a
  # missing step can be told apart from a threshold that is merely too strict.
  add_executable(owh-profile tools/profile_tool.cpp)
  add_executable(vision-replay tools/vision_replay.cpp)
  add_executable(owh-teach WIN32 tools/teach_recorder.cpp)
  add_executable(owh-vision-calibrate tools/local_vision.cpp)
  add_executable(learning-tests tests/learning_tests.cpp)
  add_test(NAME learning COMMAND learning-tests)
  add_test(NAME teaching-recorder COMMAND owh-teach --self-test --output ${CMAKE_CURRENT_BINARY_DIR}/teaching-self-test)
  set_tests_properties(learning teaching-recorder PROPERTIES TIMEOUT 30)
  configure_file(scripts/ocr-worker.ps1 ocr-worker.ps1 COPYONLY)
  add_test(NAME platform COMMAND platform-tests)
  add_test(NAME recorder COMMAND recorder-tests)
  add_test(NAME cn-recovery COMMAND cn-replay)
  add_test(NAME ocr-integration COMMAND vision-replay --self-test)
  set_tests_properties(ocr-integration PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
  foreach(target automation-protocol-tests overwatch-helper overwatch-helper-simple owh-demo owh-recorder platform-tests background-input-tests recovery-lifecycle-tests neac-cleanup-tests ui-layout-tests simple-ui-tests recorder-tests core-tests vision-tests deathmatch-tests cn-replay owh-profile vision-replay embedded-ocr-tests owh-teach owh-vision-calibrate visual-calibration-tests learning-tests)
    target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
    if(MSVC)
      target_compile_options(${target} PRIVATE /utf-8 /W4 /EHsc)
    else()
      target_link_options(${target} PRIVATE -static)
    endif()
  endforeach()
  add_custom_command(TARGET overwatch-helper POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    $<TARGET_FILE:overwatch-helper> $<TARGET_FILE_DIR:overwatch-helper>/overwatch-helper-dev.exe)
  foreach(target overwatch-helper vision-replay)
    add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      ${CMAKE_CURRENT_SOURCE_DIR}/scripts/ocr-worker.ps1 $<TARGET_FILE_DIR:${target}>/ocr-worker.ps1)
  endforeach()
  if(NOT MSVC)
    target_link_options(overwatch-helper PRIVATE -municode)
    target_link_options(overwatch-helper-simple PRIVATE -municode)
    target_link_options(owh-demo PRIVATE -municode)
    target_link_options(owh-recorder PRIVATE -municode)
    target_link_options(owh-teach PRIVATE -municode)
  endif()
  target_link_libraries(overwatch-helper-simple PRIVATE user32 gdi32 shell32 advapi32 comdlg32 winhttp ole32 dwmapi comctl32)
  target_link_libraries(overwatch-helper PRIVATE user32 gdi32 shell32 advapi32 comdlg32 winhttp ole32 dwmapi comctl32)
  target_link_libraries(simple-ui-tests PRIVATE user32 gdi32 shell32 advapi32 comdlg32 winhttp ole32 dwmapi comctl32)
  target_link_libraries(ui-layout-tests PRIVATE user32 gdi32 shell32 advapi32 comdlg32 winhttp ole32 dwmapi comctl32)
  target_link_libraries(owh-demo PRIVATE user32 gdi32)
  target_link_libraries(background-input-tests PRIVATE user32 gdi32 shell32 advapi32 comdlg32 winhttp ole32 dwmapi)
  target_link_libraries(platform-tests PRIVATE user32 gdi32 shell32 advapi32 comdlg32 winhttp ole32 dwmapi)
  target_link_libraries(recovery-lifecycle-tests PRIVATE user32 gdi32 shell32 advapi32 comdlg32 winhttp ole32 dwmapi)
  target_link_libraries(neac-cleanup-tests PRIVATE user32 gdi32 shell32 advapi32 comdlg32 winhttp ole32 dwmapi)
  foreach(target owh-recorder recorder-tests cn-replay owh-profile vision-replay embedded-ocr-tests owh-teach owh-vision-calibrate learning-tests)
    target_link_libraries(${target} PRIVATE user32 gdi32 shell32 advapi32 comdlg32 winhttp ole32
                          iphlpapi ws2_32 psapi wevtapi gdiplus uuid dwmapi)
  endforeach()
endif()
